async function fxSha(text){const b=await crypto.subtle.digest('SHA-256',new TextEncoder().encode(text.split('\r\n').join('\n')));return[...new Uint8Array(b)].map(x=>x.toString(16).padStart(2,'0')).join('');}
async function fxCheckApp(k){
  const a=fxApp(k);fxOut(k,'Comparing this device with the website and with GitHub&hellip;');
  if(!(window.crypto&&crypto.subtle)){fxOut(k,'<span class="fx-no">This browser cannot compute fingerprints here (it needs a secure page).</span>');return;}
  const rows=[];let stale=0,disagree=0,unknown=0;
  for(const f of a.files){
    const url=FX_SITE+f;let dev=null,pg=null,raw=null;
    try{if(window.caches){const m=await caches.match(url);if(m)dev=await fxSha(await m.text());}}catch(e){}
    try{const r=await fetch(url,{cache:'no-store'});if(r.ok)pg=await fxSha(await r.text());}catch(e){}
    try{const r=await fetch(FX_RAW+f+'?x='+Date.now(),{cache:'no-store'});if(r.ok)raw=await fxSha(await r.text());}catch(e){}
    let s='';
    if(!pg){s='<span class="fx-no">could not load from the website</span>';unknown++;}
    else if(!raw){s='not on GitHub, so it cannot be compared';unknown++;}
    else if(pg!==raw){s='<span class="fx-no">website and GitHub differ</span> (may be mid-update; try again in a few minutes)';disagree++;}
    else if(dev&&dev!==pg){s='<span class="fx-no">your device keeps an OLDER or different saved copy</span> (use Repair)';stale++;}
    else s='<span class="fx-ok">same</span>'+(dev?' (website, GitHub and your saved copy agree)':' (website and GitHub agree; nothing saved on this device)');
    rows.push('<tr><td>'+f+'</td><td>'+s+'</td></tr>');
  }
  const verdict=disagree?'<b class="fx-no">The website and GitHub disagree.</b> If this lasts more than a few minutes, please tell us: it could mean something was changed.':stale?'<b class="fx-no">Your device is not running the newest copy.</b> Press Repair.':unknown?'<b>Could not fully compare.</b>':'<b class="fx-ok">All match.</b>';
  fxOut(k,'<table class="fx-table" style="margin:8px 0"><tr><th>File</th><th>Result</th></tr>'+rows.join('')+'</table>'+verdict);
}

function renderAllPages(){
  const prev=document.querySelector('#sidebarPages .page-nav.active'),prevPage=prev&&/^page-[0-9]+$/.test(prev.id)?+prev.id.slice(5):1;
  const prevTabEl=document.querySelector('#detail-1 .ch-tab-content.active'),prevTab=prevTabEl?+prevTabEl.id.split('-').pop():1;
  const nav=document.getElementById('sidebarPages');
  const main=document.getElementById('mainArea');
  const pages=fxPages();
  nav.innerHTML=pages.map((p,i)=>`<div class="page-nav${i===0?' active':''}" id="page-${p.id}" onclick="openPage(${p.id})" data-tip="${p.name}">
      <div class="page-icon">${p.icon}</div>
      <div class="page-info"><div class="page-name">${p.name}</div><div class="page-sub">${p.sub}</div></div>
    </div>`).join('');
  main.innerHTML=pages.map((p,i)=>`<div class="detail${i===0?' active':''}" id="detail-${p.id}" style="flex-direction:column">${p.html}</div>`).join('');
  initPageTooltips('sidebarPages','right');
  fxRefresh();
  fxProbe();
  if(document.getElementById('fxKind'))fxSendRefresh();
  fxFillPacket();
  if(prevTab>1)fxShowTab(prevTab);
  if(prevPage>1)openPage(prevPage);
}
