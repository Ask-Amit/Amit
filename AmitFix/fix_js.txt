function fxToggle(k){
  if(fxChosen.has(k))fxChosen.delete(k);else fxChosen.add(k);
  fxRefresh();
}
function fxRefresh(){
  FX_ORDER.forEach(k=>{const el=document.getElementById('fxc-'+k);if(el)el.classList.toggle('on',fxChosen.has(k));});
  const n=fxChosen.size,bar=document.getElementById('fxBarCount');
  if(!bar)return;
  const list=[...fxChosen].sort((a,b)=>FX_ORDER.indexOf(a)-FX_ORDER.indexOf(b)).map(k=>FX_ACTIONS[k].name);
  bar.innerHTML=n?`<b>Your call sheet (${n}):</b> ${list.join(' &middot; ')}`:`<b>Your call sheet is empty.</b> Tap the cards you want, then call it.`;
  const dl=document.getElementById('fxBarGo');if(dl)dl.style.opacity=n?1:.4;
}
function fxPick(key){
  const s=FX_SYMPTOMS[key];if(!s)return;
  fxChosen.clear();s.pick.forEach(k=>fxChosen.add(k));
  fxShowTab(s.tab);fxRefresh();
}
function fxShowTab(n){
  const d=document.getElementById('detail-1');if(!d)return;
  d.querySelectorAll('.ch-tab-content').forEach(c=>c.classList.remove('active'));
  d.querySelectorAll('.ch-tab-btn').forEach((b,i)=>b.classList.toggle('active',i===n-1));
  const c=document.getElementById('ch-tab-1-'+n);if(c)c.classList.add('active');
}
function fxOrdered(){return [...fxChosen].sort((a,b)=>FX_ORDER.indexOf(a)-FX_ORDER.indexOf(b));}

async function fxGetScript(){
  for(const u of ['Amit_Fix.ps1',FX_RAW+'Amit_Fix.ps1']){
    try{const r=await fetch(u);if(r.ok){const t=await r.text();if(t.length>1500)return t;}}catch(e){}
  }
  return null;
}
async function fxDownload(){
  const acts=fxOrdered();
  if(!acts.length){alert('Your call sheet is empty. Tap one or more cards first.');return;}
  if(fxPhone){alert('These fixes run on the Windows computer where Amit is installed. Open this page on that computer (the Send It page can email or text you the link).');return;}
  const ps=await fxGetScript();
  if(!ps){alert('Could not load the fix program. Check your connection and try again.');return;}
  const hdr=['@echo off','setlocal','title Amit Fix','set "AMIT_FIX_ACTIONS='+acts.join(',')+'"',
    'powershell -NoProfile -ExecutionPolicy Bypass -Command "$t=[IO.File]::ReadAllText(\'%~f0\'); $i=$t.IndexOf(\'#PS\'+\'-START#\'); & ([scriptblock]::Create($t.Substring($i+10))) -Actions $env:AMIT_FIX_ACTIONS -Pause"',
    'exit /b','#PS-START#'];
  const body=hdr.join('\r\n')+'\r\n'+ps.replace(/\r?\n/g,'\r\n');
  const a=document.createElement('a');
  a.href=URL.createObjectURL(new Blob([body],{type:'application/octet-stream'}));
  a.download='Amit_Fix.cmd';document.body.appendChild(a);a.click();a.remove();
  fxAfterDownload();
}

// ---- looks at the person's own situation (device, and whether Amit's helper answers on this computer) ----
async function fxProbe(){
  const el=document.getElementById('fxProbe');if(!el)return;
  let msg='';
  if(fxPhone){
    msg='You are on a <b>phone or tablet</b>. Amit\'s helper only lives on a Windows computer, so the fixes below are for that computer. Use the <b>Send It</b> page to send yourself the link, and the last tab here to clear what Amit saved in <i>this</i> browser.';
  }else if(!fxWindows){
    msg='You are not on Windows. Amit\'s helper is Windows-only, so nothing of it can be on this computer. If you only want to clear what Amit saved in this browser, use the <b>Browser &amp; phone</b> tab.';
  }else{
    msg='Checking whether Amit\'s helper is running on this computer&hellip;';
    el.innerHTML=msg;
    try{
      const c=new AbortController();setTimeout(()=>c.abort(),2500);
      const r=await fetch('http://localhost:8710/api/whisper/status',{signal:c.signal});
      const j=await r.json();
      msg='Amit\'s helper <b class="fx-ok">is running</b> on this computer'+(j&&j.state?' (voice typing: <b>'+j.state+'</b>)':'')+'.'+(fxVersion?' Version '+fxVersion+'.':'');
    }catch(e){
      msg='I can\'t see Amit\'s helper on this computer. Either it is not installed, or it is not running right now. "Look first" will tell you for sure, and "Restart" or "Fresh start" will bring it back.';
    }
  }
  el.innerHTML=msg;
}

function fxClearBrowser(){
  if(!confirm('Sign out and clear what Amit saved in this browser on this device? (Your online account is not deleted.)'))return;
  let n=0;
  try{Object.keys(localStorage).forEach(k=>{if(/amit|^sb-/i.test(k)){localStorage.removeItem(k);n++;}});}catch(e){}
  try{sessionStorage.clear();}catch(e){}
  try{if(db)db.auth.signOut();}catch(e){}
  const el=document.getElementById('fxBrowserDone');
  if(el)el.innerHTML=`<span class="fx-ok">Done.</span> Signed out and cleared ${n} saved item${n===1?'':'s'} for Amit in this browser.`;
}

function fxCopy(id,btn){
  const el=document.getElementById(id);if(!el)return;
  const t=el.value||el.textContent;
  const done=()=>{const o=btn.textContent;btn.textContent='Copied';setTimeout(()=>btn.textContent=o,1400);};
  if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(t).then(done,()=>{el.select&&el.select();document.execCommand('copy');done();});}
  else{el.select&&el.select();document.execCommand('copy');done();}
}

function renderAllPages(){
  const prev=document.querySelector('#sidebarPages .page-nav.active'),prevPage=prev&&/^page-[0-9]+$/.test(prev.id)?+prev.id.slice(5):1;
  const prevTabEl=document.querySelector('#detail-1 .ch-tab-content.active'),prevTab=prevTabEl?+prevTabEl.id.split('-').pop():1;
  const nav=document.getElementById('sidebarPages');
  const main=document.getElementById('mainArea');
  const pages=fxPages();
  nav.innerHTML=pages.map((p,i)=>`<div class="page-nav${i===0?' active':''}" id="page-${p.id}" onclick="openPage(${p.id})" data-tip="${p.name}">
      <div class="page-icon">${p.icon}</div>
      <div class="page-info"><div class="page-name">${p.name}</div><div class="page-sub">${p.sub}</div></div>
    </div>`).join('');
  main.innerHTML=pages.map((p,i)=>`<div class="detail${i===0?' active':''}" id="detail-${p.id}" style="flex-direction:column">${p.html}</div>`).join('');
  initPageTooltips('sidebarPages','right');
  fxRefresh();
  fxProbe();
  if(document.getElementById('fxKind'))fxSendRefresh();
  if(prevTab>1)fxShowTab(prevTab);
  if(prevPage>1)openPage(prevPage);
}
